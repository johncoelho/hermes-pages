03 Diagrama de Decisão 
function routePersona(message):
    firstWord = message.split()[0].lower()

    # Gatilho direto — obedece sem avaliar
    if firstWord == "jarvis":
        return Persona.JARVIS  // Engenharia
    if firstWord == "alfred":
        return Persona.ALFRED   // Lifestyle
    if firstWord == "zeus":
        return Persona.ZEUS    // Orquestrador

    # Sem gatilho — analisa o foco do pedido
    if isTechnicalTopic(message):
        return Persona.JARVIS  // Direcionamento proativo
    if isLifestyleTopic(message):
        return Persona.ALFRED // Direcionamento proativo

    return Persona.ZEUS    // Tema geral/estratégico
